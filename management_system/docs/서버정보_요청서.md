## [B] 서버에서 직접 실행해 결과를 붙여주셔도 됩니다 (PowerShell, 관리자)

```powershell
# OS·사양
Get-ComputerInfo -Property OsName, OsVersion, CsTotalPhysicalMemory, CsNumberOfLogicalProcessors
Get-PSDrive -PSProvider FileSystem | Select Name, Used, Free
# 사용 중인 포트와 프로세스 (다른 엔진 포트 확인)
Get-NetTCPConnection -State Listen | Select LocalAddress, LocalPort, @{n='Process';e={(Get-Process -Id $_.OwningProcess).ProcessName}} | Sort LocalPort
# 등록된 서비스 중 직접 설치한 것
Get-Service | Where-Object {$_.StartType -eq 'Automatic'} | Select Name, DisplayName, Status
# 런타임
python --version; py -0p; git --version; node --version
# NAS 접근 확인 (경로는 실제 값으로)
Test-Path \\192.168.0.10\BMS
# 예약 작업
Get-ScheduledTask | Where-Object {$_.TaskPath -eq '\'} | Select TaskName, State
```
